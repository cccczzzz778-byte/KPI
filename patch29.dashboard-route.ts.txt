import { commissions, criteria, defaultInstitutions, type CommissionKey } from "@/lib/kpi-data";
import { getKpiDatabase } from "@/lib/netlify-db";

export const runtime = "nodejs";
export const dynamic = "force-dynamic";

type PeriodMode = "daily" | "monthly" | "60d" | "directions";
type DailyAggregate = {
  institutionId: string;
  evaluationDate: string;
  ijro: number;
  birlamchi: number;
  statsionar: number;
  raqam: number;
  moliya: number;
  assessedCount: number;
};

type Institution = {
  id: string;
  name: string;
  district: string;
  type: string;
  active: number;
};

const KEYS: CommissionKey[] = ["ijro", "birlamchi", "statsionar", "raqam", "moliya"];
const criterionIds = Object.fromEntries(
  KEYS.map((key) => [key, criteria.filter((item) => item.commission === key).map((item) => item.id)]),
) as Record<CommissionKey, string[]>;
const criterionCounts = Object.fromEntries(KEYS.map((key) => [key, criterionIds[key].length])) as Record<CommissionKey, number>;

function tashkentDate() {
  const parts = new Intl.DateTimeFormat("en-US", { timeZone: "Asia/Tashkent", year: "numeric", month: "2-digit", day: "2-digit" }).formatToParts(new Date());
  const get = (type: string) => parts.find((part) => part.type === type)?.value ?? "";
  return `${get("year")}-${get("month")}-${get("day")}`;
}

function validDate(value: string) {
  return /^\d{4}-\d{2}-\d{2}$/.test(value) && !Number.isNaN(Date.parse(`${value}T12:00:00Z`));
}

function shiftDate(value: string, days: number) {
  const date = new Date(`${value}T12:00:00Z`);
  date.setUTCDate(date.getUTCDate() + days);
  return date.toISOString().slice(0, 10);
}

function monthStart(value: string) {
  return `${value.slice(0, 7)}-01`;
}

function normalize(raw: number, key: CommissionKey) {
  const max = Math.max(1, criterionCounts[key] * 2);
  return Number(((Number(raw || 0) / max) * 20).toFixed(1));
}

function emptyParts() {
  return { ijro: 0, birlamchi: 0, statsionar: 0, raqam: 0, moliya: 0 } as Record<CommissionKey, number>;
}

function normalizedParts(row?: DailyAggregate) {
  const parts = emptyParts();
  if (!row) return parts;
  for (const key of KEYS) parts[key] = normalize(Number(row[key] || 0), key);
  return parts;
}

function totalParts(parts: Record<CommissionKey, number>) {
  return Number(KEYS.reduce((sum, key) => sum + Number(parts[key] || 0), 0).toFixed(1));
}

export async function GET(request: Request) {
  try {
    const url = new URL(request.url);
    const currentDate = tashkentDate();
    const requestedDate = String(url.searchParams.get("date") || "");
    const selectedDate = validDate(requestedDate) ? requestedDate : currentDate;
    const requestedPeriod = String(url.searchParams.get("period") || "daily");
    const period: PeriodMode = (["daily", "monthly", "60d", "directions"] as string[]).includes(requestedPeriod)
      ? (requestedPeriod as PeriodMode)
      : "daily";
    const selectedMonth = selectedDate.slice(0, 7);
    const selectedMonthStart = monthStart(selectedDate);
    const sixtyStart = shiftDate(selectedDate, -59);
    const trendStart = shiftDate(selectedDate, -6);
    const rangeStart = [selectedMonthStart, sixtyStart, trendStart].sort()[0];

    const db = getKpiDatabase();
    const [institutionResult, dailyResult, fileResult] = await Promise.all([
      db.pool.query(`
        SELECT id, name, district, type, active
        FROM institutions
        ORDER BY LOWER(name)
      `),
      db.pool.query(
        `SELECT institution_id AS "institutionId",
                TO_CHAR(evaluation_date, 'YYYY-MM-DD') AS "evaluationDate",
                SUM(CASE WHEN criterion_id = ANY($1::text[]) THEN score ELSE 0 END)::int AS ijro,
                SUM(CASE WHEN criterion_id = ANY($2::text[]) THEN score ELSE 0 END)::int AS birlamchi,
                SUM(CASE WHEN criterion_id = ANY($3::text[]) THEN score ELSE 0 END)::int AS statsionar,
                SUM(CASE WHEN criterion_id = ANY($4::text[]) THEN score ELSE 0 END)::int AS raqam,
                SUM(CASE WHEN criterion_id = ANY($5::text[]) THEN score ELSE 0 END)::int AS moliya,
                COUNT(DISTINCT criterion_id)::int AS "assessedCount"
           FROM daily_evaluations
          WHERE evaluation_date BETWEEN $6::date AND $7::date
          GROUP BY institution_id, evaluation_date
          ORDER BY evaluation_date ASC`,
        [criterionIds.ijro, criterionIds.birlamchi, criterionIds.statsionar, criterionIds.raqam, criterionIds.moliya, rangeStart, selectedDate],
      ),
      db.pool.query(
        `SELECT institution_id AS "institutionId",
                TO_CHAR(submission_date, 'YYYY-MM-DD') AS "submissionDate",
                COUNT(*)::int AS count
           FROM attachments
          WHERE source IN ('institution', 'institution_submission')
            AND submission_date BETWEEN $1::date AND $2::date
          GROUP BY institution_id, submission_date`,
        [rangeStart, selectedDate],
      ),
    ]);

    const institutionMap = new Map<string, Institution>(
      defaultInstitutions.map((item) => [item.id, { ...item, active: 1 } as Institution]),
    );
    for (const row of institutionResult.rows) institutionMap.set(String(row.id), row as Institution);
    const institutions = Array.from(institutionMap.values())
      .filter((item) => Number(item.active) === 1)
      .sort((a, b) => String(a.name).localeCompare(String(b.name), "uz"));

    const aggregates = dailyResult.rows.map((row) => ({
      institutionId: String(row.institutionId),
      evaluationDate: String(row.evaluationDate),
      ijro: Number(row.ijro || 0),
      birlamchi: Number(row.birlamchi || 0),
      statsionar: Number(row.statsionar || 0),
      raqam: Number(row.raqam || 0),
      moliya: Number(row.moliya || 0),
      assessedCount: Number(row.assessedCount || 0),
    })) as DailyAggregate[];

    const aggregateByInstitution = new Map<string, DailyAggregate[]>();
    for (const row of aggregates) {
      const list = aggregateByInstitution.get(row.institutionId) ?? [];
      list.push(row);
      aggregateByInstitution.set(row.institutionId, list);
    }

    const fileCountMap = new Map<string, number>();
    for (const row of fileResult.rows) {
      fileCountMap.set(`${String(row.institutionId)}:${String(row.submissionDate)}`, Number(row.count || 0));
    }
    const filesForRange = (institutionId: string, start: string, end: string) => {
      let total = 0;
      for (const [key, count] of fileCountMap.entries()) {
        const splitAt = key.lastIndexOf(":");
        const id = key.slice(0, splitAt);
        const date = key.slice(splitAt + 1);
        if (id === institutionId && date >= start && date <= end) total += count;
      }
      return total;
    };

    const buildDaily = (institution: Institution, date: string) => {
      const aggregate = (aggregateByInstitution.get(institution.id) ?? []).find((item) => item.evaluationDate === date);
      const parts = normalizedParts(aggregate);
      return {
        id: institution.id,
        name: institution.name,
        district: institution.district,
        type: institution.type,
        parts,
        total: totalParts(parts),
        maxTotal: 100,
        evaluated: Boolean(aggregate?.assessedCount),
        assessedCount: Number(aggregate?.assessedCount || 0),
        fileCount: filesForRange(institution.id, date, date),
        daysEvaluated: aggregate?.assessedCount ? 1 : 0,
      };
    };

    const buildMonthly = (institution: Institution) => {
      const rows = (aggregateByInstitution.get(institution.id) ?? []).filter((item) => item.evaluationDate >= selectedMonthStart && item.evaluationDate <= selectedDate && item.assessedCount > 0);
      const parts = emptyParts();
      if (rows.length) {
        for (const key of KEYS) {
          parts[key] = Number((rows.reduce((sum, row) => sum + normalize(Number(row[key] || 0), key), 0) / rows.length).toFixed(1));
        }
      }
      return {
        id: institution.id,
        name: institution.name,
        district: institution.district,
        type: institution.type,
        parts,
        total: totalParts(parts),
        maxTotal: 100,
        evaluated: rows.length > 0,
        assessedCount: rows.reduce((sum, row) => sum + row.assessedCount, 0),
        fileCount: filesForRange(institution.id, selectedMonthStart, selectedDate),
        daysEvaluated: rows.length,
      };
    };

    const build60 = (institution: Institution) => {
      const rows = (aggregateByInstitution.get(institution.id) ?? []).filter((item) => item.evaluationDate >= sixtyStart && item.evaluationDate <= selectedDate && item.assessedCount > 0);
      const parts = emptyParts();
      for (const key of KEYS) parts[key] = Number(rows.reduce((sum, row) => sum + normalize(Number(row[key] || 0), key), 0).toFixed(1));
      return {
        id: institution.id,
        name: institution.name,
        district: institution.district,
        type: institution.type,
        parts,
        total: totalParts(parts),
        maxTotal: 6000,
        evaluated: rows.length > 0,
        assessedCount: rows.reduce((sum, row) => sum + row.assessedCount, 0),
        fileCount: filesForRange(institution.id, sixtyStart, selectedDate),
        daysEvaluated: rows.length,
      };
    };

    const rows = institutions.map((institution) => {
      if (period === "monthly") return buildMonthly(institution);
      if (period === "60d") return build60(institution);
      return buildDaily(institution, selectedDate);
    }).sort((a, b) => b.total - a.total || a.name.localeCompare(b.name, "uz"));

    const evaluatedRows = rows.filter((row) => row.evaluated);
    const averageScore = evaluatedRows.length
      ? Number((evaluatedRows.reduce((sum, row) => sum + row.total, 0) / evaluatedRows.length).toFixed(1))
      : 0;
    const periodFileCount = rows.reduce((sum, row) => sum + row.fileCount, 0);

    const trend = Array.from({ length: 7 }, (_, index) => shiftDate(selectedDate, index - 6)).map((date) => {
      const dayRows = institutions.map((institution) => buildDaily(institution, date)).filter((row) => row.evaluated);
      return {
        date,
        label: `${date.slice(8, 10)}.${date.slice(5, 7)}`,
        averageKpi: dayRows.length ? Number((dayRows.reduce((sum, row) => sum + row.total, 0) / dayRows.length).toFixed(1)) : 0,
        evaluatedCount: dayRows.length,
      };
    });

    const todayRows = institutions.map((institution) => buildDaily(institution, selectedDate)).filter((row) => row.evaluated);
    const directionAverages = commissions.map((commission) => ({
      id: commission.id,
      short: commission.short,
      title: commission.title,
      value: todayRows.length
        ? Number((todayRows.reduce((sum, row) => sum + Number(row.parts[commission.id] || 0), 0) / todayRows.length).toFixed(1))
        : 0,
      max: 20,
    }));

    return Response.json({
      currentDate,
      selectedDate,
      selectedMonth,
      period,
      totalCriteria: criteria.length,
      totalInstitutions: institutions.length,
      evaluatedCount: evaluatedRows.length,
      unevaluatedCount: Math.max(0, institutions.length - evaluatedRows.length),
      fileCount: periodFileCount,
      averageScore,
      maxAverageScore: period === "60d" ? 6000 : 100,
      rows,
      trend,
      directionAverages,
    }, { headers: { "cache-control": "no-store, no-cache, must-revalidate" } });
  } catch (error) {
    console.error("public dashboard failed", error);
    const message = error instanceof Error ? error.message : "Dashboard ma’lumotlarini yuklab bo‘lmadi.";
    return Response.json({ error: message }, { status: 500, headers: { "cache-control": "no-store" } });
  }
}
